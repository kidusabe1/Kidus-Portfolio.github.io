def generate_poisson_spikes(rate, T):
    spike_times = []
    t = 0

    while t < T:
        # Generate the next spike interval
        interval = -np.log(np.random.rand()) / rate # Exponentially distributed inter-spike interval
        t += interval
        if t < T:
            spike_times.append(t)

    return spike_times


# piece wise rate function to model the refractory and recovery period of a neuron after a spike
def rate_function(t, r=55):
    # return 0 if t is within 5ms after a spike, then return a recovery function for 6ms, then return the original rate
    if t < 0.005:
        return 0
    elif t < 0.011:
        return r * (t - 0.005) / 0.006
    else:
        return r

# apply the rate function to account for the refractory period
# rejection method from the already generated homegenous poisson neuron spike times,
#  we will adjust the spike times based on the rate function to account for the refractory period.
# If a spike occurs within 5ms of the last spike, we will skip it.
# If it occurs between 5ms and 11ms, we will adjust its timing based on the rate function.
adjusted_spike_times = []
for spike in spike_times:
    if len(adjusted_spike_times) == 0:
        adjusted_spike_times.append(spike)
    else:
        last_spike = adjusted_spike_times[-1]
        interval = spike - last_spike
        if interval < 0.005:
            continue  # skip this spike due to refractory period
        elif interval < 0.011:
            # adjust probabilty of this spike based on the rate function
            if np.random.rand() < rate_function(interval) / 55:
                adjusted_spike_times.append(spike)
        else:
            adjusted_spike_times.append(spike)

# Survivor Function

ISI_count_vec, bin_edges = np.histogram(ISI_adjusted, bins=500, density=True)

pmf_vec = ISI_count_vec / np.sum(ISI_count_vec)

cdf_vec = np.cumsum(pmf_vec)
survivor_function = 1 - cdf_vec

# plot both cumulative and survivor function in subplots
axs = plt.figure(figsize=(12, 6)).subplots(1, 2)
axs[0].plot(bin_edges[:-1], cdf_vec, label='Cumulative Distribution Function (CDF)')
axs[0].set_title('Cumulative Distribution Function (CDF)')
axs[0].set_xlabel('ISI (s)')
